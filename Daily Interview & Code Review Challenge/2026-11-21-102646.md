## Bối cảnh

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddDbContext<TimesheetDbContext>(o =>
    o.UseSqlServer(builder.Configuration.GetConnectionString("Timesheet")));
builder.Services.AddControllers();
var app = builder.Build();
using (var scope = app.Services.CreateScope())
{
    var db = scope.ServiceProvider.GetRequiredService<TimesheetDbContext>();
    db.Database.Migrate();
}
app.MapControllers();
app.Run();

[ApiController]
[Route("api/timesheets")]
public sealed class TimesheetController : ControllerBase
{
    private readonly TimesheetDbContext _db;
    public TimesheetController(TimesheetDbContext db) => _db = db;

    [HttpPost("{id:int}/approve")]
    public async Task<IActionResult> Approve(int id)
    {
        var sheet = await _db.Timesheets.FindAsync(id);
        if (sheet is null) return NotFound();
        sheet.ApprovalState = "Approved";
        await _db.SaveChangesAsync();
        return Ok(new { sheet.Id, sheet.ApprovalState });
    }

    [HttpGet("{id:int}")]
    public async Task<IActionResult> Get(int id)
    {
        var sheet = await _db.Timesheets.AsNoTracking()
            .SingleOrDefaultAsync(x => x.Id == id);
        return sheet is null ? NotFound() :
            Ok(new { sheet.Id, sheet.ApprovalState });
    }
}

## Kết quả code review tham khảo

### [Critical] — Drop cột phá vỡ legacy readers/writers

**Vấn đề:** Xóa `IsApproved` trong lần deploy đầu.  
**Tại sao xảy ra:** Node cũ và ứng dụng legacy vẫn dùng schema cũ.  
**Ảnh hưởng production:** SQL exceptions, HTTP 500, rollback binary thất bại.  
**Cách sửa:** Expand–migrate–contract; chỉ drop khi không còn consumer cũ.

### [Critical] — Backfill sai trạng thái lịch sử

**Vấn đề:** `ApprovalState` mặc định Pending cho mọi bản ghi.  
**Tại sao xảy ra:** Không chuyển dữ liệu từ `IsApproved`.  
**Ảnh hưởng production:** Timesheet đã duyệt bị coi là chưa duyệt.  
**Cách sửa:** Backfill theo mapping được xác nhận, reconciliation trước/sau chuyển đổi.

### [High] — Authorization không thể hiện trong endpoint

**Vấn đề:** Không có authentication/resource authorization trong đoạn code.  
**Tại sao xảy ra:** Controller tin rằng người gọi có quyền.  
**Ảnh hưởng production:** Unauthorized approvals và truy cập timesheet.  
**Cách sửa:** Enforce authentication, policy và kiểm tra quyền trên timesheet; xác minh global policies nếu có.

### [High] — DDL thực thi trong startup

**Vấn đề:** Mỗi application instance gọi `Migrate()`.  
**Tại sao xảy ra:** Schema deployment bị gắn vào app lifecycle.  
**Ảnh hưởng production:** Startup failure, tranh chấp triển khai, quyền DDL rộng.  
**Cách sửa:** Migration bundle/script trong release stage độc lập, reviewed và monitored. Không giả định EF Core migration locking thay thế deployment orchestration.

### [High] — Dual-version writes không tương thích

**Vấn đề:** Binary mới chỉ ghi `ApprovalState`; binary cũ ghi `IsApproved`.  
**Tại sao xảy ra:** Chưa có compatibility protocol.  
**Ảnh hưởng production:** Các nodes hiển thị trạng thái khác nhau.  
**Cách sửa:** Dual-write hoặc compatibility adapter, reconciliation, xác định thời điểm cutover.

### [Medium] — Không kiểm tra transition/concurrency

**Vấn đề:** Approval ghi trực tiếp string, không kiểm tra trạng thái hoặc competing updates.  
**Tại sao xảy ra:** Thiếu domain transition và concurrency token.  
**Ảnh hưởng production:** Lost update, workflow sai.  
**Cách sửa:** Validate transitions, audit, rowversion và xử lý `DbUpdateConcurrencyException`.

### [Medium] — Thiếu cancellation propagation

**Vấn đề:** EF Core I/O không nhận request cancellation.  
**Tại sao xảy ra:** Không truyền `CancellationToken`.  
**Ảnh hưởng production:** Tốn tài nguyên sau khi request bị hủy.  
**Cách sửa:** Propagate token; không coi cancellation là transaction rollback.

### [Medium] — Thiếu release telemetry

**Vấn đề:** Không có schema-version checks hoặc rollout gates.  
**Tại sao xảy ra:** Release chỉ dựa vào startup thành công.  
**Ảnh hưởng production:** Phát hiện regression muộn.  
**Cách sửa:** Canary, smoke tests, structured logs, SQL metrics và consistency alerts.

## Phiên bản đề xuất

Giả định **expanded schema đã tồn tại**, giữ `IsApproved`, thêm `ApprovalState` nullable và `RowVersion`; backfill được triển khai riêng. `IApprovalPermission` là abstraction nghiệp vụ phải được hiện thực và kiểm thử, không phải API built-in.

```csharp
// Migration: chạy qua release pipeline, KHÔNG gọi Database.Migrate() khi startup.
// Expand: add nullable ApprovalState; giữ IsApproved; backfill có kiểm chứng.
// Configure RowVersion as EF Core concurrency token.

[Authorize]
[ApiController]
[Route("api/timesheets")]
public sealed class TimesheetController : ControllerBase
{
    private readonly TimesheetDbContext _db;
    private readonly IApprovalPermission _permission;

    public TimesheetController(TimesheetDbContext db, IApprovalPermission permission)
        => (_db, _permission) = (db, permission);

    [HttpPost("{id:int}/approve")]
    public async Task<IActionResult> Approve(int id, CancellationToken ct)
    {
        var sheet = await _db.Timesheets
            .SingleOrDefaultAsync(x => x.Id == id, ct);
        if (sheet is null) return NotFound();
        if (!await _permission.CanApproveAsync(User, sheet, ct)) return Forbid();
        if (sheet.ApprovalState == "Approved") return NoContent();
        if (sheet.ApprovalState is not ("Pending" or null))
            return Conflict(new ProblemDetails { Title = "Invalid transition" });

        sheet.ApprovalState = "Approved";
        sheet.IsApproved = true; // compatibility window
        sheet.ApprovedAtUtc = DateTime.UtcNow;
        try { await _db.SaveChangesAsync(ct); }
        catch (DbUpdateConcurrencyException)
        {
            return Conflict(new ProblemDetails { Title = "Concurrent update" });
        }
        return NoContent();
    }
}
```

Triển khai theo thứ tự: kiểm kê consumers và backup → expand schema → backfill theo batches → deploy compatible readers/writers → canary/verify → chuyển read path bằng feature flag → theo dõi consistency → contract/drop cột ở release riêng. Nếu legacy app ghi SQL trực tiếp, dual-write trong API mới **không đủ**; cần bổ sung compatibility tại boundary phù hợp.

## Giải thích thiết kế sau khi sửa

DDL thuộc release orchestration; application identity không cần quyền sửa schema. Compatibility window giữ cả hai representations. Resource authorization và concurrency token bảo vệ nghiệp vụ. Audit cần ghi transactionally với approval hoặc qua outbox nếu phát event.

## Trade-offs

Expand–contract tốn nhiều release và reconciliation; dual-write có thể divergence. Rowversion làm lộ conflicts thay vì âm thầm overwrite. Migration một bước chỉ khả thi khi có downtime được chấp thuận và mọi consumer đồng thời chuyển đổi.

## Cách kiểm thử

1. **Integration:** binary cũ/mới cùng đọc ghi expanded schema.
2. **Migration:** dữ liệu approved/pending được backfill đúng, rollback binary vẫn chạy.
3. **Security:** người không có quyền không duyệt/xem timesheet phòng ban khác.
4. **Concurrency:** hai approvers đồng thời không mất update hoặc audit.
5. **Deployment:** canary trên một IIS node, synthetic requests qua cả hai nodes và rollback drill.

## Quan sát trong production

Application Insights: 5xx theo node/version, dependency failures, p95 approve; SQL Server: migration duration, blocking/deadlocks, schema-version errors; structured logs: approval transition, actor ID phù hợp privacy, concurrency conflicts; alert trên divergence giữa hai cột và tỷ lệ 500 sau rollout.

## Câu hỏi phỏng vấn mở rộng

1. Nếu ứng dụng legacy không thể thay đổi trong sáu tháng thì sao?
2. Backfill bảng 100 triệu dòng mà không khóa lâu thế nào?
3. Làm sao đảm bảo approval và audit event không lệch nhau?

<script setup lang="ts">
import { onMounted, ref, watch } from "vue";

type Result = { id: string; title: string; snippet: string };
const query = ref("");
const results = ref<Result[]>([]);
const loading = ref(false);
const error = ref("");
const selected = ref<string[]>([]);

async function search() {
  loading.value = true;
  error.value = "";
  const response = await fetch("/api/search?q=" + query.value);
  const data = await response.json();
  results.value = data.items;
  loading.value = false;
}

watch(query, () => {
  search();
});

onMounted(() => {
  search();
});

function openResult(id: string) {
  selected.value.push(id);
  window.location.href = "/documents/" + id;
}
</script>

<template>
  <section>
    <input v-model="query" placeholder="Search documents" />
    <p v-if="loading">Loading...</p>
    <p v-if="error">{{ error }}</p>
    <ul>
      <li v-for="item in results" :key="item.id">
        <a href="#" @click.prevent="openResult(item.id)">
          {{ item.title }}
        </a>
        <div v-html="item.snippet"></div>
      </li>
    </ul>
  </section>
</template>

## Kết quả code review tham khảo

### [Critical] — Untrusted HTML trong v-html

**Vấn đề:** Render CMS snippet thành HTML.  
**Tại sao xảy ra:** Bỏ qua escaping mặc định của Vue.  
**Ảnh hưởng production:** Stored XSS trên origin portal.  
**Cách sửa:** Interpolate text; nếu cần highlight, dùng text + ranges hoặc sanitizer đã được kiểm định và CSP defense-in-depth.

### [High] — Stale response ghi đè state

**Vấn đề:** Request cũ hoàn thành muộn có thể cập nhật `results`.  
**Tại sao xảy ra:** Không có cancellation hoặc request identity guard.  
**Ảnh hưởng production:** Kết quả sai keyword, chọn nhầm tài liệu.  
**Cách sửa:** Abort request trước và kiểm tra sequence ID trước khi update state.

### [High] — Không xử lý lỗi HTTP/network

**Vấn đề:** Không check `response.ok`, thiếu try/catch/finally.  
**Tại sao xảy ra:** `fetch` resolve bình thường với HTTP 4xx/5xx.  
**Ảnh hưởng production:** UI loading mãi, unhandled rejection, response lỗi được coi là dữ liệu.  
**Cách sửa:** Check status, validate JSON, reset loading trong finally, xử lý auth/rate-limit theo contract.

### [Medium] — Request amplification

**Vấn đề:** Một request cho mỗi ký tự.  
**Tại sao xảy ra:** Watcher gọi trực tiếp.  
**Ảnh hưởng production:** Tăng backend load, latency và chi phí search.  
**Cách sửa:** Debounce, giới hạn query và rate limiting server-side.

### [Medium] — URL query không encode

**Vấn đề:** Nối chuỗi query trực tiếp.  
**Tại sao xảy ra:** Không dùng URLSearchParams.  
**Ảnh hưởng production:** Ký tự đặc biệt phá query semantics.  
**Cách sửa:** URLSearchParams.

### [Medium] — Loading và result state không nhất quán

**Vấn đề:** Request cũ set loading false trong khi request mới chạy.  
**Tại sao xảy ra:** Shared state không gắn với active request.  
**Ảnh hưởng production:** UX gây hiểu lầm.  
**Cách sửa:** Sequence guard cả loading, error và results; clear khi input rỗng.

### [Low] — Navigation semantics kém

**Vấn đề:** `href="#"`, state selected thừa, imperative redirect.  
**Tại sao xảy ra:** Dùng click handler thay link thực.  
**Ảnh hưởng production:** Accessibility, open-new-tab và browser history kém.  
**Cách sửa:** Link thật, encode route segment, server-side resource authorization.

### [Low] — Tin tuyệt đối response shape

**Vấn đề:** `data.items` không validate, không giới hạn số kết quả.  
**Tại sao xảy ra:** Không xử lý API contract drift.  
**Ảnh hưởng production:** UI crash hoặc render list quá lớn.  
**Cách sửa:** Runtime validation và pagination/limit phía server.

## Phiên bản đề xuất

```vue
<script setup lang="ts">
import { ref, watch } from "vue";

type Result = { id: string; title: string; snippet: string };
const query = ref("");
const results = ref<Result[]>([]);
const loading = ref(false);
const error = ref("");
let sequence = 0;

watch(query, (value, _old, onCleanup) => {
  const current = ++sequence;
  const controller = new AbortController();
  results.value = [];
  loading.value = false;
  error.value = "";
  const term = value.trim();
  if (term.length < 2) return;

  const timer = setTimeout(async () => {
    loading.value = true;
    try {
      const params = new URLSearchParams({ q: term, limit: "20" });
      const response = await fetch(`/api/search?${params}`, {
        signal: controller.signal,
        credentials: "same-origin"
      });
      if (!response.ok) throw new Error(`HTTP ${response.status}`);
      const data: unknown = await response.json();
      if (!data || typeof data !== "object" || !("items" in data)
          || !Array.isArray(data.items)) throw new Error("Invalid response");
      const items = data.items as unknown[];
      if (!items.every((x): x is Result =>
        !!x && typeof x === "object" &&
        typeof (x as Result).id === "string" &&
        typeof (x as Result).title === "string" &&
        typeof (x as Result).snippet === "string"))
        throw new Error("Invalid item");
      if (current === sequence) results.value = items.slice(0, 20);
    } catch {
      if (current === sequence && !controller.signal.aborted)
        error.value = "Không tải được kết quả.";
    } finally {
      if (current === sequence) loading.value = false;
    }
  }, 300);
  onCleanup(() => {
    clearTimeout(timer);
    controller.abort();
  });
});
</script>

<template>
  <section>
    <label for="search">Tìm tài liệu</label>
    <input id="search" v-model="query" />
    <p v-if="loading" role="status">Đang tìm...</p>
    <p v-if="error" role="alert">{{ error }}</p>
    <ul>
      <li v-for="item in results" :key="item.id">
        <a :href="'/documents/' + encodeURIComponent(item.id)">
          {{ item.title }}
        </a>
        <div>{{ item.snippet }}</div>
      </li>
    </ul>
  </section>
</template>
```

## Giải thích thiết kế sau khi sửa

Watcher cleanup hủy timer và request trước khi chạy search mới hoặc unmount. Sequence ID ngăn stale completion cập nhật state. Debounce giảm request bursts; URLSearchParams encode query; interpolation escape HTML. API vẫn cần authorization, rate limiting và pagination.

## Trade-offs

Debounce thêm khoảng 300 ms latency nhưng giảm tải. Abort không đảm bảo server đã dừng xử lý. Runtime validation tăng code; schema library có thể phù hợp khi API lớn. Nếu cần highlight rich text, text + highlight ranges thường dễ kiểm soát hơn raw HTML.

## Cách kiểm thử

1. **Race:** response query cũ về sau query mới không được overwrite.
2. **Security:** malicious HTML snippet hiển thị dạng text, không thực thi.
3. **Failure-path:** 401/429/500, malformed JSON, network error đều kết thúc loading.
4. **Component:** fake timers chứng minh debounce và cleanup khi unmount.
5. **Playwright E2E:** Unicode, ký tự `&`, navigation và authorization ở backend.

## Quan sát trong production

Browser telemetry: request count/search session, aborted requests, JS errors, search-to-render latency. Backend: p95/p99, HTTP 4xx/5xx/429, dependency latency, rate-limit events; không log raw sensitive search terms. Dùng correlation IDs để nối browser và Application Insights/Kibana; alert nếu error rate tăng sau release.
